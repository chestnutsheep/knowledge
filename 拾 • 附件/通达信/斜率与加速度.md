  - 价格斜率( ROC )与量能变化
	```
	零界:0,COLORGRAY,LINETHICK1;
	出货:8,LINETHICK1,COLORRED;

	吞筹:-8,LINETHICK1,COLORRED;
	
	ROC:=100*(C-REF(C,1))/REF(C,1),LINETHICK2;
	MAROC:=MA(ROC,5);
	
	VROC:=10*(V-REF(V,1))/REF(V,1),LINETHICK2;
	MAVROC:=MA(VROC,5);
	
	一阶离差:ROC-VROC,COLORYELLOW;
	
	DRAWBAND(IF(一阶离差>0,MIN(一阶离差,出货),MAX(一阶离差,吞筹)),RGB(239,130,165),0,RGB(43,174,133));
	```

 - 信号
	 ```
	ROC:=100*(C-REF(C,1))/REF(C,1);
	VROC:=5*(V-REF(V,1))/REF(V,1);
	一阶离差:=ROC-VROC;
	
	D2C:=ROC-REF(ROC,1);
	D2V:=VROC-REF(VROC,1);
	二阶离差:=D2C-D2V;
	
	0,COLORGRAY,LINETHICK1;
	出货:8,LINETHICK1,COLORRED;
	警:4,DOTLINE,COLORYELLOW,LINETHICK1;
	戒:-4,DOTLINE,COLORYELLOW,LINETHICK1;
	吞筹:-8,LINETHICK1,COLORGREEN;
	
	一阶:一阶离差,LINETHICK1,COLORWHITE;
	二阶:二阶离差*2,LINETHICK1,COLORLIGRAY;
	
	STICKLINE(1,10.3,9.7,2,0),COLORGRAY;
	T1绿:=BETWEEN(一阶离差,-3.99,3.99);
	T1黄:=(一阶离差>=4 AND 一阶离差<8) OR (一阶离差<=-4 AND 一阶离差>-8);
	T1红:=一阶离差>=8 OR 一阶离差<=-8;
	STICKLINE(T1绿,10.3,9.7,2,0),COLORGREEN;
	STICKLINE(T1黄,10.3,9.7,2,0),COLORYELLOW;
	STICKLINE(T1红,10.3,9.7,2,0),COLORRED;
	
	STICKLINE(1,0.3,-0.3,2,0),COLORGRAY;
	T3绿:=T1绿 AND BETWEEN(二阶离差*2,-1.99,1.99);
	T3红:=T1红 OR 二阶离差*2>=4 OR 二阶离差*2<=-4;
	T3黄:=NOT(T3绿) AND NOT(T3红);
	STICKLINE(T3绿,0.3,-0.3,2,0),COLORGREEN;
	STICKLINE(T3黄,0.3,-0.3,2,0),COLORYELLOW;
	STICKLINE(T3红,0.3,-0.3,2,0),COLORRED;
	
	STICKLINE(1,-9.7,-10.3,2,0),COLORGRAY;
	T2绿:=BETWEEN(二阶离差*2,-1.99,1.99);
	T2黄:=(二阶离差*2>=2 AND 二阶离差*2<4) OR (二阶离差*2<=-2 AND 二阶离差*2>-4);
	T2红:=二阶离差*2>=4 OR 二阶离差*2<=-4;
	STICKLINE(T2绿,-9.7,-10.3,2,0),COLORGREEN;
	STICKLINE(T2黄,-9.7,-10.3,2,0),COLORYELLOW;
	STICKLINE(T2红,-9.7,-10.3,2,0),COLORRED;
    ```
